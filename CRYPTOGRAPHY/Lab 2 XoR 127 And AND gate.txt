#include <stdio.h>

int main() {
    char *str = "World";

    printf("Original String: %s\n", str);

    printf("After AND with 127: ");
    for(int i = 0; str[i] != '\0'; i++) {
        printf("%c", str[i] & 127);
    }

    printf("\n");

    printf("After XOR with 127: ");
    for(int i = 0; str[i] != '\0'; i++) {
        printf("%c", str[i] ^ 127);
    }

    printf("\n");

    return 0;
}
