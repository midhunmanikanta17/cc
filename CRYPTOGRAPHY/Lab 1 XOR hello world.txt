#include <stdio.h>

int main() {
    char str[100];

    printf("Enter the string: ");
    scanf("%99s", str);

    printf("After XOR with 0: ");

    for(int i = 0; str[i] != '\0'; i++) {
        char result = str[i] ^ 0;
        printf("%c", result);
    }

    printf("\n");

    return 0;
}
